Panduan Konfigurasi Supabase & Skema PostgreSQL

1. Skema Tabel PostgreSQL (Supabase)

Skema lengkap mencakup chart_of_accounts, journal_headers, journal_lines, dan audit_logs.

-- 1. Chart of Accounts (COA)
CREATE TABLE chart_of_accounts (
    account_code VARCHAR(20) PRIMARY KEY,
    account_name VARCHAR(100) NOT NULL,
    account_type VARCHAR(30) NOT NULL, -- Asset, Liability, Expense, dll
    is_active BOOLEAN DEFAULT TRUE
);

-- 2. Journal Headers
CREATE TABLE journal_headers (
    voucher_no VARCHAR(50) PRIMARY KEY,
    voucher_date DATE NOT NULL,
    description TEXT NOT NULL,
    status VARCHAR(20) CHECK (status IN ('DRAFT', 'POSTED', 'REVERSED')) DEFAULT 'DRAFT',
    total_debit NUMERIC(15,2) NOT NULL,
    total_credit NUMERIC(15,2) NOT NULL,
    hash_checksum VARCHAR(64) NOT NULL,
    created_by VARCHAR(50) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. Journal Lines
CREATE TABLE journal_lines (
    line_id SERIAL PRIMARY KEY,
    voucher_no VARCHAR(50) REFERENCES journal_headers(voucher_no) ON DELETE CASCADE,
    account_code VARCHAR(20) REFERENCES chart_of_accounts(account_code),
    debit NUMERIC(15,2) DEFAULT 0.00,
    credit NUMERIC(15,2) DEFAULT 0.00,
    CONSTRAINT chk_positive_amounts CHECK (debit >= 0 AND credit >= 0)
);

-- 4. Audit Logs (Immutable)
CREATE TABLE audit_logs (
    log_id SERIAL PRIMARY KEY,
    voucher_no VARCHAR(50),
    action VARCHAR(50) NOT NULL,
    performed_by VARCHAR(50) NOT NULL,
    performed_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    details TEXT
);

2. Fungsi RPC PostgreSQL (`create_voucher` & `post_voucher`) 

-- Fungsi RPC untuk membuat voucher baru dengan validasi ketat di Database
CREATE OR REPLACE FUNCTION create_voucher(
    p_voucher_no VARCHAR,
    p_voucher_date DATE,
    p_description TEXT,
    p_created_by VARCHAR,
    p_lines JSONB,
    p_checksum VARCHAR
) RETURNS JSON AS $$
DECLARE
    v_total_debit NUMERIC := 0;
    v_total_credit NUMERIC := 0;
    v_line JSONB;
    v_line_debit NUMERIC;
    v_line_credit NUMERIC;
    v_count INT := 0;
BEGIN
    -- Validasi keberadaan nomor voucher
    IF EXISTS (SELECT 1 FROM journal_headers WHERE voucher_no = p_voucher_no) THEN
        RAISE EXCEPTION 'Nomor voucher % sudah terdaftar.', p_voucher_no;
    END IF;

    -- Hitung dan validasi baris dari JSONB
    FOR v_line IN SELECT * FROM jsonb_array_elements(p_lines)
    LOOP
        v_count := v_count + 1;
        v_line_debit := (v_line->>'debit')::NUMERIC;
        v_line_credit := (v_line->>'credit')::NUMERIC;

        -- Aturan: Tidak boleh negatif dan hanya salah satu yang terisi positif per baris
        IF v_line_debit < 0 OR v_line_credit < 0 THEN
            RAISE EXCEPTION 'Nominal tidak boleh bernilai negatif.';
        END IF;
        IF v_line_debit > 0 AND v_line_credit > 0 THEN
            RAISE EXCEPTION 'Baris jurnal tidak boleh memiliki debit dan kredit secara bersamaan.';
        END IF;

        v_total_debit := v_total_debit + v_line_debit;
        v_total_credit := v_total_credit + v_line_credit;
    END LOOP;

    IF v_count < 2 THEN
        RAISE EXCEPTION 'Minimal harus terdapat 2 baris jurnal.';
    END IF;

    IF v_total_debit <> v_total_credit THEN
        RAISE EXCEPTION 'Total debit (%) tidak seimbang dengan total kredit (%).', v_total_debit, v_total_credit;
    END IF;

    -- Insert ke header
    INSERT INTO journal_headers (voucher_no, voucher_date, description, status, total_debit, total_credit, hash_checksum, created_by)
    VALUES (p_voucher_no, p_voucher_date, p_description, 'DRAFT', v_total_debit, v_total_credit, p_checksum, p_created_by);

    -- Insert ke lines
    FOR v_line IN SELECT * FROM jsonb_array_elements(p_lines)
    LOOP
        INSERT INTO journal_lines (voucher_no, account_code, debit, credit)
        VALUES (
            p_voucher_no,
            v_line->>'account_code',
            (v_line->>'debit')::NUMERIC,
            (v_line->>'credit')::NUMERIC
        );
    END LOOP;

    -- Catat audit log
    INSERT INTO audit_logs (voucher_no, action, performed_by, details)
    VALUES (p_voucher_no, 'CREATE_VOUCHER_DRAFT', p_created_by, 'Voucher berhasil dibuat sebagai Draft.');

    RETURN json_build_object('success', true, 'voucher_no', p_voucher_no);
END;
$$ LANGUAGE plpgsql;

3. Alur Validasi & Integritas SHA-256

Sistem menerapkan validasi berlapis (*Two-Tier Validation*). Validasi pertama berjalan di sisi *frontend* (JavaScript) untuk memberikan umpan balik instan kepada pengguna. Validasi kedua dan utama dijalankan di sisi *database* melalui fungsi RPC PostgreSQL.

Setiap kali voucher disimpan, string data utama (Nomor Voucher + Tanggal + Total Debit + Daftar Akun) dikonversi menggunakan algoritma SHA-256 checksum untuk memastikan integritas data. Jika jurnal telah berstatus POSTED, pembaruan atau penghapusan langsung ditolak oleh pemicu basis data (*Database Trigger*), mewajibkan pembuatan jurnal pembalik (*Reversing Entry*) jika koreksi diperlukan.
